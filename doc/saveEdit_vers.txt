const saveEdit = () => {
    const newText = input.value.trim() || '[ Empty ]';
    if (newText !== todo.body) {
        if (confirm(`Update this todo?\n\nOld: "${todo.body}"\nNew: "${newText}"`)) {
            todo.body = newText;
            this.setTodosInStorage();
            this.toHTML(); // Re-render to reflect update
        } else {
            this.toHTML(); // Just re-render to restore the original view
        }
    } else {
        this.toHTML(); // No change, just re-render
    }
};
